目录管理页面实现指南

在当前 Vue 3 与 Element Plus 工程中实现目录管理页。按顺序复制完整文件和局部修改，即可从后台侧边栏进入页面，选择书籍、编排目录并预览文章。

新增一个 API 模块、两个子组件和一个页面，沿用现有的登录、请求封装和管理布局。每个代码块的标题会标明目标文件与放置位置。

1. 从选择书籍到打开文件

登录后进入 /admin/catalogues。初次进入只加载天体，选择天体后列出该天体下的书籍。选书后展示书籍自身的显隐、全书排序进度、左侧目录树和右侧当前目录的直接子项。选择目录浏览下一层，点击文件在当前页预览 Markdown。

页面动作管理 API页面取值或动作
进入页面GET /v1/planet/list复用 getBodyList()，以字符串 ID 为选项值，用 getBodyLabel(name) 显示中文名称。
选择天体POST /v1/book/options发送 {planet: {id}}，使用 data.books，包含该天体下的隐藏书籍。
选择书籍或同步目录POST /v1/catalogue/tree发送 {book: {id}}。读取 data.book.catalogues。同级的 children 已由服务端排好。
在抽屉中保存排序POST /v1/catalogue/reorder同一份请求提交父节点 ID、打开抽屉时的分组与最终分组。成功后重读树。
切换显隐POST /v1/catalogue/display发送 {catalogue: {id, display_status}}，code === 200 即写入成功。用最新目录数据同步开关。
点击文件POST /v1/article/preview发送 {catalogue: {id}}。data.article 提供文章主键、标题和 Markdown 正文。

浏览器通过现有的 request 实例发起请求。开发环境会把 /api/v1/… 代理到管理后端，示例函数只传 /v1/…。该实例已负责添加 Bearer token、判断业务码、解包 data 并处理登录失效。

3. 封装页面请求

新建以下模块。getBookOptions() 可供目录管理和按天体选书的其他页面共用，天体列表沿用 src/api/body.js。新函数都接收业务参数、返回解包后的业务值。接口抛出的业务错误会保留给页面处理。

新建文件src/api/catalogue.js
import request from '@/utils/request'
import {isRecord} from '@/utils/typeGuard'
import {readRequiredStringField, readSafeIntegerField, readStringField} from '@/utils/fileReader'
import {BOOK_SORT_MAX} from '@/constants/book'

/** @typedef {import('@/constants/book').BookDisplayStatus} DisplayStatus */
/**
 * @typedef {Object} BookOption
 * @property {string} id 书籍 ID.
 * @property {string} title 书名.
 * @property {number} sort 书籍排序.
 * @property {DisplayStatus} display_status 书籍自身的显隐.
 */
/**
 * @typedef {Object} CatalogueNode
 * @property {string} id 目录节点 ID.
 * @property {string} parent_id 直接父节点的 ID, 根节点为 "0".
 * @property {string} name 目录名或文件名.
 * @property {'folder' | 'file'} type 节点类型.
 * @property {number | null} sort 序号, null 表示待排.
 * @property {DisplayStatus} display_status 节点自身的显隐.
 * @property {CatalogueNode[]} children 已按阅读顺序排列的直接子节点.
 */
/** @typedef {BookOption & {catalogues: CatalogueNode[]}} CatalogueBook */
/** @typedef {{sorted_children: {id: string}[], unsorted_children: {id: string}[]}} OrderGroups */
/** @typedef {{catalogue: {parent_id: string, expected: OrderGroups, target: OrderGroups}}} ReorderRequest */
/** @typedef {{id: string, title: string, content: string}} ArticlePreview */

/**
 * 读取书籍或目录的显示状态.
 * @param {Record<string, unknown>} record 解包后的记录.
 * @returns {DisplayStatus} 服务端保存的状态.
 */
function readDisplayStatus(record) {
    const status = readRequiredStringField(record, 'display_status')
    if (status !== 'visible' && status !== 'hidden') {
        throw new Error('显示状态无效')
    }
    return status
}

/**
 * 读取书籍下拉选项与目录树共有的书籍字段.
 * @param {unknown} value 接口返回的书籍.
 * @returns {BookOption} 书籍摘要.
 */
function readBookOption(value) {
    if (isRecord(value) === false) {
        throw new Error('书籍记录格式异常')
    }
    return {
        id: readRequiredStringField(value, 'id'),
        title: readStringField(value, 'title'),
        // 读取时保留历史书籍的序号 0.
        sort: readSafeIntegerField(value, 'sort', 0, BOOK_SORT_MAX),
        display_status: readDisplayStatus(value),
    }
}

/**
 * 读取后端已组装的节点, 保留各层 children 的顺序.
 * @param {unknown} value 目录树中的一个节点.
 * @returns {CatalogueNode} 可交给页面的节点.
 */
function readCatalogueNode(value) {
    if (isRecord(value) === false || Array.isArray(value.children) === false) {
        throw new Error('目录节点缺少 children 数组')
    }
    const type = readRequiredStringField(value, 'type')
    if (type !== 'folder' && type !== 'file') {
        throw new Error('目录节点类型无效')
    }
    const parentId = readRequiredStringField(value, 'parent_id')
    let sort = null
    if (parentId === '0') {
        sort = readSafeIntegerField(value, 'sort', 0, 0)
    } else if (value.sort !== null) {
        sort = readSafeIntegerField(value, 'sort', 1, Number.MAX_SAFE_INTEGER)
    }
    return {
        id: readRequiredStringField(value, 'id'),
        parent_id: parentId,
        name: readStringField(value, 'name'),
        type,
        sort,
        display_status: readDisplayStatus(value),
        children: value.children.map(readCatalogueNode),
    }
}

/**
 * 读取选中天体下的全部书籍选项, 包括隐藏书籍.
 * @param {string} planetId 天体 ID.
 * @returns {Promise<BookOption[]>} 服务端排序后的书籍选项.
 */
async function getBookOptions(planetId) {
    /** @type {unknown} request 已解包的 data. */
    const data = await request.post('/v1/book/options', {planet: {id: planetId}})
    if (isRecord(data) === false || Array.isArray(data.books) === false) {
        throw new Error('书籍选项响应缺少 books 数组')
    }
    return data.books.map(readBookOption)
}

/**
 * 读取一本书和它的完整目录树.
 * @param {string} bookId 书籍 ID.
 * @returns {Promise<CatalogueBook>} 尚未导入目录时 catalogues 是空数组.
 */
async function getCatalogueTree(bookId) {
    /** @type {unknown} request 已解包的 data. */
    const data = await request.post('/v1/catalogue/tree', {book: {id: bookId}})
    if (isRecord(data) === false || isRecord(data.book) === false ||
        Array.isArray(data.book.catalogues) === false) {
        throw new Error('目录树响应缺少 book.catalogues 数组')
    }
    const book = readBookOption(data.book)
    if (book.id !== bookId) {
        throw new Error('目录树响应中的书籍 ID 与请求不一致')
    }
    return {...book, catalogues: data.book.catalogues.map(readCatalogueNode)}
}

/**
 * 保存父目录下的阅读顺序. 两个分组均包含该层全部直接子项.
 * @param {ReorderRequest} payload 已含外层 catalogue 的完整请求体.
 * @returns {Promise<void>} 排序事务已提交.
 */
async function saveCatalogueOrder(payload) {
    await request.post('/v1/catalogue/reorder', payload)
}

/**
 * 设置单个节点的目标状态, 由后端处理级联隐藏.
 * @param {string} id 目录节点 ID.
 * @param {DisplayStatus} status 目标显隐状态.
 * @returns {Promise<void>} 显隐事务已提交.
 */
async function setCatalogueDisplay(id, status) {
    await request.post('/v1/catalogue/display', {catalogue: {id, display_status: status}})
}

/**
 * 用文件节点 ID 读取文章. 响应内的 id 是文章主键.
 * @param {string} nodeId 所点击的文件节点 ID.
 * @returns {Promise<ArticlePreview>} 标题与保留换行的 Markdown 正文.
 */
async function getArticlePreview(nodeId) {
    /** @type {unknown} request 已解包的 data. */
    const data = await request.post('/v1/article/preview', {catalogue: {id: nodeId}})
    if (isRecord(data) === false || isRecord(data.article) === false) {
        throw new Error('预览响应缺少 article 对象')
    }
    return {
        id: readRequiredStringField(data.article, 'id'),
        title: readStringField(data.article, 'title'),
        content: readStringField(data.article, 'content'),
    }
}

export {getBookOptions, getCatalogueTree, saveCatalogueOrder, setCatalogueDisplay, getArticlePreview}

所有 ID 按字符串传递。目录根序号 0 合法，非根节点的 null 表示待排。书籍选项和目录树中的 display_status 属于不同记录，界面须分别读取。空的 catalogues 表示该书暂未导入目录。

4. 编排一个父目录的直接子项

新建排序抽屉。它在打开时复制本层条目，将 id 分成有顺序的 sorted_children 和按成员比较的 unsorted_children。两组覆盖本层全部直接子项。管理员在草稿中加入、上移、下移或移回待排，首次打开时的 expected 始终代表原分组。

新建文件src/components/CatalogueOrderDrawer.vue
<template>
    <el-drawer
        :model-value="modelValue" title="编排同级节点" size="min(560px, 100%)"
        :show-close="!busy" :close-on-press-escape="!busy" :close-on-click-modal="!busy"
        :before-close="beforeClose" @update:model-value="changeOpen" @opened="focusCandidate"
    >
        <div ref="listArea" :aria-busy="busy">
            <p>{{ parentName }}</p>
            <p>将节点加入阅读顺序后, 可用上下箭头调整位置. 保存时, 可见目录移回待排区会隐藏整棵子树, 后代保留原序号.</p>
            <h3>阅读顺序 ({{ ranked.length }})</h3>
            <p v-if="ranked.length === 0">请从待排区加入节点.</p>
            <ol v-else class="draft-list">
                <li v-for="(node, index) in ranked" :key="node.id" :data-node="node.id" tabindex="-1">
                    <span class="node-name">{{ index + 1 }}. {{ node.name }}</span>
                    <div class="row-actions">
                        <button type="button" :disabled="busy || index === 0"
                                :aria-label="'上移 ' + node.name" @click="moveAt(index, -1)">↑</button>
                        <button type="button" :disabled="busy || index === ranked.length - 1"
                                :aria-label="'下移 ' + node.name" @click="moveAt(index, 1)">↓</button>
                        <button type="button" :disabled="busy" @click="removeAt(index)">
                            {{ node.sort === null ? '撤回加入' : '移回待排' }}
                        </button>
                    </div>
                </li>
            </ol>
            <h3>待排区 ({{ pending.length }})</h3>
            <p v-if="pending.length === 0">本级节点已全部加入阅读顺序.</p>
            <ul v-else class="draft-list">
                <li v-for="node in pending" :key="node.id" :data-node="node.id" tabindex="-1">
                    <span class="node-name">{{ node.name }}</span>
                    <button type="button" :disabled="busy" @click="addNode(node)">加入</button>
                </li>
            </ul>
        </div>
        <template #footer>
            <el-button :disabled="busy" @click="changeOpen(false)">取消</el-button>
            <el-button type="primary" :loading="busy" :disabled="busy || !changed" @click="submit">
                保存排序
            </el-button>
        </template>
    </el-drawer>
</template>

<script setup>
import {computed, nextTick, ref, watch} from 'vue'

/** @typedef {import('@/api/catalogue').CatalogueNode} CatalogueNode */
/** @typedef {Pick<CatalogueNode, 'id' | 'name' | 'sort'>} DraftNode */
/** @typedef {import('@/api/catalogue').OrderGroups} OrderGroups */

/** 抽屉显示状态, 当前父目录, 可选的快捷编排项和提交状态. */
const props = defineProps({
    modelValue: {type: Boolean, required: true},
    parent: {type: /** @type {import('vue').PropType<CatalogueNode | null>} */ (Object), default: null},
    candidateId: {type: String, default: ''},
    busy: {type: Boolean, default: false},
})
/** 父页面执行 API 写入和目录刷新. */
const emit = defineEmits(['update:modelValue', 'save'])
/** @type {import('vue').Ref<HTMLElement | null>} 抽屉内的列表区域. */
const listArea = ref(null)
/** 打开时的父目录 ID. */
const parentId = ref('')
/** 打开时的父目录名称. */
const parentName = ref('')
/** @type {import('vue').Ref<DraftNode[]>} 本层记录的副本. */
const siblings = ref([])
/** @type {import('vue').Ref<DraftNode[]>} 按阅读顺序排列的草稿. */
const ranked = ref([])
/** @type {OrderGroups} 打开时的快照, 编辑草稿期间保留原值. */
let expected = {sorted_children: [], unsorted_children: []}
/** 待排项按中文名称自然排序. */
const collator = new Intl.Collator('zh-CN', {numeric: true})
/** 剩余节点形成待排组, 保证每个直接子项仅出现一次. */
const pending = computed(function () {
    const rankedIds = new Set(ranked.value.map(node => node.id))
    return siblings.value.filter(node => !rankedIds.has(node.id)).sort(function (a, b) {
        return collator.compare(a.name, b.name) || collator.compare(a.id, b.id)
    })
})
/** 成员变化或已排项换位均允许保存. */
const changed = computed(function () {
    const original = expected.sorted_children
    return ranked.value.length !== original.length ||
        ranked.value.some((node, index) => node.id !== original[index].id)
})

/**
 * 生成 API 要求的两份完整子项数组.
 * @returns {OrderGroups} ID 分组.
 */
function groups() {
    return {
        sorted_children: ranked.value.map(node => ({id: node.id})),
        unsorted_children: pending.value.map(node => ({id: node.id})),
    }
}

/** 每次打开时取得新的快照, 将快捷编入操作应用到草稿. */
function startDraft() {
    const parent = props.parent
    if (parent === null) return
    parentId.value = parent.id
    parentName.value = parent.name
    siblings.value = parent.children.map(node => ({id: node.id, name: node.name, sort: node.sort}))
    ranked.value = siblings.value.filter(node => node.sort !== null)
    expected = groups()
    const candidate = siblings.value.find(node => node.id === props.candidateId)
    if (candidate && candidate.sort === null) ranked.value.push(candidate)
}
watch(() => props.modelValue, function (open) {
    if (open) startDraft()
}, {immediate: true})

/**
 * 将焦点放在移动后的节点, 支持键盘连续操作.
 * @param {string} id 节点 ID.
 */
async function focusRow(id) {
    await nextTick()
    if (listArea.value === null || id === '') return
    const row = listArea.value.querySelector('[data-node="' + CSS.escape(id) + '"]')
    if (row instanceof HTMLElement) row.focus()
}

/** 从表格行进入时滚动到相应节点. */
function focusCandidate() {
    focusRow(props.candidateId)
}

/**
 * 加入阅读顺序.
 * @param {DraftNode} node 待排节点.
 */
function addNode(node) {
    if (props.busy) return
    ranked.value.push(node)
    focusRow(node.id)
}

/**
 * 将节点移回待排组.
 * @param {number} index 当前索引.
 */
function removeAt(index) {
    if (props.busy) return
    const [node] = ranked.value.splice(index, 1)
    focusRow(node.id)
}

/**
 * 调整已排节点的位置.
 * @param {number} index 当前索引.
 * @param {number} delta 移动距离.
 */
function moveAt(index, delta) {
    const to = index + delta
    if (props.busy || to < 0 || to >= ranked.value.length) return
    const [node] = ranked.value.splice(index, 1)
    ranked.value.splice(to, 0, node)
    focusRow(node.id)
}

/**
 * 写入期间阻止关闭.
 * @param {Function} done Element Plus 提供的关闭回调.
 */
function beforeClose(done) {
    if (!props.busy) done()
}

/**
 * 通知父页面切换抽屉.
 * @param {boolean} open 期望的打开状态.
 */
function changeOpen(open) {
    if (!props.busy) emit('update:modelValue', open)
}

/** 发送完整请求体, 父页面根据服务端结果决定是否关闭. */
function submit() {
    if (props.busy || !changed.value) return
    emit('save', {catalogue: {parent_id: parentId.value, expected, target: groups()}})
}
</script>

例如本层已排第三章、第一章，第二章待排。把第三章移回待排，再加入第二章后，expected.sorted_children 仍为第三章、第一章，target.sorted_children 变为第一章、第二章，第三章进入 target.unsorted_children。只有点击「保存排序」才发送写请求。返回 20302 时先读取最新目录，让管理员重新确认。

5. 从文件节点打开文章预览

新建预览对话框。点击文件时以该节点的 id 请求，返回的 article.id 属于文章记录。后端读取磁盘正文并改写本地图片 URL，组件直接使用返回的 content。渲染顺序是 Markdown → HTML → DOMPurify → v-html。Marked 文档 · DOMPurify 用法

新建文件src/components/ArticlePreviewDialog.vue
<template>
    <el-dialog
        :model-value="modelValue"
        :title="title"
        width="min(860px, 94vw)"
        append-to-body
        @update:model-value="emit('update:modelValue', $event)"
    >
        <p v-if="loading" role="status">正在读取正文...</p>
        <div v-else-if="errorText !== ''" role="alert">
            <p>{{ errorText }}</p>
            <el-button @click="loadPreview">重新读取</el-button>
        </div>
        <article v-else-if="html !== ''" class="markdown" v-html="html"></article>
        <el-empty v-else description="这篇文章暂无正文" />
    </el-dialog>
</template>

<script setup>
import {computed, onBeforeUnmount, ref, watch} from 'vue'
import {marked} from 'marked'
import DOMPurify from 'dompurify'
import {getArticlePreview} from '@/api/catalogue'
import {API_CODE} from '@/constants/apiCode'
import {getApiErrorCode, getApiErrorMessage, isAuthError} from '@/utils/apiError'

/** @typedef {import('@/api/catalogue').CatalogueNode} CatalogueNode */
const props = defineProps({
    modelValue: {type: Boolean, required: true},
    node: {type: /** @type {import('vue').PropType<CatalogueNode>} */ (Object), default: null},
})
const emit = defineEmits(['update:modelValue', 'lost'])

/** 只展示本次打开时请求的正文. */
const loading = ref(false)
const html = ref('')
const errorText = ref('')
const articleTitle = ref('')
let requestNumber = 0

const title = computed(function () {
    if (articleTitle.value !== '') return articleTitle.value
    if (props.node !== null) return props.node.name
    return 'Markdown 预览'
})

/** 打开文件或点击重试时读取正文. 已关闭或已切换文件的响应会被丢弃. */
async function loadPreview() {
    const number = ++requestNumber
    const node = props.node
    html.value = ''
    errorText.value = ''
    articleTitle.value = ''
    loading.value = false
    if (!props.modelValue || node === null || node.type !== 'file') return

    loading.value = true
    try {
        const article = await getArticlePreview(node.id)
        if (number !== requestNumber) return
        articleTitle.value = article.title
        const rendered = marked.parse(article.content, {async: false, gfm: true})
        html.value = DOMPurify.sanitize(rendered, {USE_PROFILES: {html: true}})
    } catch (error) {
        if (number !== requestNumber || isAuthError(error)) return
        const code = getApiErrorCode(error)
        if (code === API_CODE.CATALOGUE_NOT_FOUND) {
            errorText.value = '该目录节点已删除'
            emit('lost')
        } else if (code === API_CODE.ARTICLE_PREVIEW_UNAVAILABLE) {
            errorText.value = '该节点的 Markdown 正文暂时无法预览'
        } else {
            errorText.value = getApiErrorMessage(error, '读取文章失败')
        }
    } finally {
        if (number === requestNumber) loading.value = false
    }
}

watch([() => props.modelValue, () => props.node?.id], loadPreview, {immediate: true})
onBeforeUnmount(function () { requestNumber++ })
</script>

6. 组合目录管理页面

新建页面组件。左栏基于后端给出的树展示可见行，右栏始终以选中目录的 children 为数据源。任何已选书都能管理，包括书籍自身被隐藏的情况。

页面状态实现时的约定
选择天体或书籍先清空下一级选项及旧目录。只接收仍属于当前选择的异步响应。
首次读取与刷新首次只展开并选中根节点。写操作后的刷新按节点 ID 恢复当前目录和展开位置，已删除的节点回退到有效目录。
全书排序进度分子为整棵树中已排序的非根节点数，分母为所有非根节点数。单独切换显隐不会改变进度。
隐藏与重新显示可见目录被隐藏，或者在保存排序时移回待排，会在服务端隐藏同书全部后代。后代保留序号。重新显示目录或重新编入排序，只显示该目录。
成功与重新同步排序和显隐写入成功都返回空对象。隐藏后重取目录树，因为重复隐藏与实际切换的响应相同。显示状态的确认只依赖显隐接口的业务结果，排序仅由抽屉发起。
写入结果未知超时或断线时先读取服务端最新状态。即便读取失败，管理员也有明确的刷新入口，可恢复后再继续编辑。

代码阅读顺序

  1. 从 selectPlanet()、selectBook() 和 loadTree() 看联动与回显。每次读取递增版本号，即使用户从 A 切到 B 又切回 A，旧响应也不会覆盖最后一次选择。
  2. indexTree() 为服务端提供的树建立 ID 索引，visibleRows 仅计算左侧已展开的行。右侧表格绑定 folder.children，保持单层。全书进度遍历全部节点，包含当前折叠的分支。
  3. openNode() 区分文件与文件夹，openOrder() 打开草稿。writeChange() 串行提交一次写请求，并处理显隐、冲突和刷新。排序成功后按原路径恢复目录，显示单个节点成功后可直接更新该开关。

<script setup>
import {computed, onBeforeUnmount, onMounted, ref} from 'vue'
import {ElMessage} from 'element-plus'
import {Document, Folder} from '@element-plus/icons-vue'
import {getBodyList} from '@/api/body'
import {getBodyLabel} from '@/utils/body'
import {getBookOptions, getCatalogueTree, saveCatalogueOrder, setCatalogueDisplay} from '@/api/catalogue'
import {API_CODE} from '@/constants/apiCode'
import {getApiErrorCode, getApiErrorMessage, isAuthError} from '@/utils/apiError'
import CatalogueOrderDrawer from '@/components/CatalogueOrderDrawer.vue'
import ArticlePreviewDialog from '@/components/ArticlePreviewDialog.vue'

/** @typedef {import('@/api/catalogue').CatalogueNode} CatalogueNode */
/** @type {import('vue').Ref<import('@/api/body').BodyRecord[]>} 天体选项. */
const planets = ref([])
/** @type {import('vue').Ref<import('@/api/catalogue').BookOption[]>} 所选天体的书籍. */
const books = ref([])
/** @type {import('vue').Ref<import('@/api/catalogue').CatalogueBook | null>} 当前已读取的书和目录树. */
const book = ref(null)
const planetId = ref('')
const bookId = ref('')
const folderId = ref('')
const fileId = ref('')
const expanded = ref(new Set())
const readError = ref('')
const reading = ref(false)
const writing = ref(false)
const orderOpen = ref(false)
const candidateId = ref('')
const previewOpen = ref(false)
/** 最近一次读取的编号. 切换选项及离开页面时, 作废较早的响应. */
let readVersion = 0
let active = true

/** @returns {Map<string, CatalogueNode>} 便于按 ID 恢复选择及获取父节点. */
function indexTree() {
    const result = new Map()
    const stack = book.value === null ? [] : [...book.value.catalogues]
    while (stack.length > 0) {
        const node = stack.pop()
        result.set(node.id, node)
        for (const child of node.children) stack.push(child)
    }
    return result
}
const index = computed(indexTree)
const folder = computed(() => index.value.get(folderId.value) || null)
const previewNode = computed(() => index.value.get(fileId.value) || null)
const selectedBook = computed(() => book.value || books.value.find(item => item.id === bookId.value))
const nonRootCount = computed(() => [...index.value.values()].filter(node => node.parent_id !== '0').length)
const sortedCount = computed(() => [...index.value.values()].filter(node => node.parent_id !== '0' && node.sort !== null).length)
const rankedChildren = computed(() => folder.value === null ? 0 : folder.value.children.filter(node => node.sort !== null).length)
const expandableIds = computed(() => [...index.value.values()].filter(node => node.children.length > 0).map(node => node.id))
const allOpen = computed(() => expandableIds.value.length > 0 && expandableIds.value.every(id => expanded.value.has(id)))
const selectionHint = computed(function () {
    if (planets.value.length === 0) return '暂无已创建的天体'
    if (planetId.value === '') return '请先选择天体'
    if (books.value.length === 0) return '该天体暂无书籍'
    return '请选择书籍'
})

/**
 * @param {CatalogueNode | null} node 路径终点.
 * @returns {CatalogueNode[]} 从根到终点的路径.
 */
function pathTo(node) {
    const path = []
    while (node) {
        path.push(node)
        node = index.value.get(node.parent_id)
    }
    return path.reverse()
}
const currentPath = computed(() => pathTo(folder.value))
/** 按后端给定的顺序生成可见行. 展开状态与节点的显示开关相互独立. */
const visibleRows = computed(function () {
    const rows = []
    function visit(node, depth) {
        rows.push({node, depth})
        if (expanded.value.has(node.id)) {
            for (const child of node.children) visit(child, depth + 1)
        }
    }
    if (book.value !== null) {
        for (const root of book.value.catalogues) visit(root, 0)
    }
    return rows
})

/** @param {string} id 手动展开或收起的节点. */
function toggleFolder(id) {
    if (expanded.value.has(id)) expanded.value.delete(id)
    else expanded.value.add(id)
}
/** 展开或收起全部有子项的节点. */
function toggleAll() {
    expanded.value = new Set(allOpen.value ? [] : expandableIds.value)
}
/** @param {CatalogueNode} node 选中的目录或待预览的文件. */
function openNode(node) {
    if (writing.value) return
    const isFile = node.type === 'file'
    fileId.value = isFile ? node.id : ''
    folderId.value = isFile ? node.parent_id : node.id
    for (const parent of currentPath.value) expanded.value.add(parent.id)
    previewOpen.value = isFile
}

/** 丢弃上一棵树的视图状态. 请求版本号由发起新读取的调用方维护. */
function clearBook() {
    book.value = null
    bookId.value = ''
    folderId.value = ''
    fileId.value = ''
    expanded.value = new Set()
    orderOpen.value = false
    previewOpen.value = false
}
/** 开始一次可作废的读取. */
function startRead() {
    readError.value = ''
    reading.value = true
    return ++readVersion
}
/** @param {number} version 此次读取编号. */
function finishRead(version) {
    if (version === readVersion) reading.value = false
}
/** 获取天体选项. */
async function loadPlanets() {
    const version = startRead()
    clearBook()
    planets.value = []
    planetId.value = ''
    books.value = []
    try {
        const result = await getBodyList()
        if (version === readVersion) planets.value = result
    } catch (error) {
        if (version === readVersion && !isAuthError(error)) readError.value = getApiErrorMessage(error, '读取天体失败')
    } finally {
        finishRead(version)
    }
}
/** @param {string} id 新选择的天体, 空串清空选择. */
async function selectPlanet(id) {
    const version = startRead()
    planetId.value = id
    books.value = []
    clearBook()
    try {
        if (id === '') return
        const result = await getBookOptions(id)
        if (version === readVersion) books.value = result
    } catch (error) {
        if (version !== readVersion || isAuthError(error)) return
        if (getApiErrorCode(error) === API_CODE.BOOK_PLANET_NOT_FOUND) {
            ElMessage.warning('所选天体已不存在, 请重新选择')
            await loadPlanets()
        } else {
            readError.value = getApiErrorMessage(error, '读取书籍选项失败')
        }
    } finally {
        finishRead(version)
    }
}
/** @param {string} id 书籍 ID, 保持为字符串. */
function selectBook(id) {
    ++readVersion
    clearBook()
    reading.value = false
    readError.value = ''
    bookId.value = id
    if (id !== '') loadTree()
}
/** 重读权威目录树, 尽量恢复选中的目录和展开位置. */
async function loadTree() {
    const id = bookId.value
    if (id === '') return
    const version = startRead()
    const previousPath = currentPath.value.map(node => node.id).reverse()
    const hadTree = book.value !== null && book.value.catalogues.length > 0
    try {
        const result = await getCatalogueTree(id)
        if (version !== readVersion) return
        book.value = result
        const root = result.catalogues[0]
        expanded.value = hadTree
            ? new Set([...expanded.value].filter(key => index.value.has(key)))
            : new Set(root ? [root.id] : [])
        const nearest = previousPath.find(key => index.value.get(key)?.type === 'folder')
        folderId.value = nearest || (root ? root.id : '')
        if (!index.value.has(fileId.value)) {
            fileId.value = ''
            previewOpen.value = false
        }
    } catch (error) {
        if (version !== readVersion || isAuthError(error)) return
        if (getApiErrorCode(error) === API_CODE.BOOK_NOT_FOUND) {
            ElMessage.warning('书籍已不存在, 请重新选择')
            await selectPlanet(planetId.value)
        } else {
            readError.value = getApiErrorMessage(error, '目录读取失败, 请重新读取')
        }
    } finally {
        finishRead(version)
    }
}
/** 按当前选择重试失败的读取. */
function retryRead() {
    if (bookId.value !== '') return loadTree()
    if (planetId.value !== '') return selectPlanet(planetId.value)
    return loadPlanets()
}
/** @param {string} id 表格行的 ID, 从标题进入时使用空串. */
function openOrder(id) {
    if (writing.value || reading.value || folder.value === null) return
    candidateId.value = id
    orderOpen.value = true
}

/** 在单次写入结束前锁定上下文. 排序或隐藏成功后重新读取整棵树.
 * @param {() => Promise<void>} send 发出一次写请求.
 * @param {boolean} isOrder 是否从排序抽屉提交.
 * @param {CatalogueNode} [shownNode] 仅显示此节点时可直接更新开关.
 */
async function writeChange(send, isOrder, shownNode) {
    if (writing.value || reading.value || readError.value !== '') return
    writing.value = true
    try {
        await send()
        if (!active) return
        if (isOrder) orderOpen.value = false
        if (shownNode) shownNode.display_status = 'visible'
        else await loadTree()
    } catch (error) {
        if (!active || isAuthError(error)) return
        const code = getApiErrorCode(error)
        if (code === API_CODE.VALIDATION_ERROR) {
            ElMessage.error(getApiErrorMessage(error, '请核对提交内容'))
            return
        }
        orderOpen.value = false
        if (code === API_CODE.CATALOGUE_ORDER_CHANGED) ElMessage.warning('同级目录已变化, 请核对新顺序后重新编排')
        else if (code === API_CODE.CATALOGUE_SORT_REQUIRED) ElMessage.warning('请先为该节点排序')
        else if (code === API_CODE.CATALOGUE_NOT_FOUND) ElMessage.warning('节点已不存在, 正在刷新目录')
        else if (code === API_CODE.BOOK_NOT_FOUND) ElMessage.warning('书籍已不存在, 请重新选择')
        else ElMessage.warning(getApiErrorMessage(error, '请重新读取目录以确认保存结果'))
        if (code === API_CODE.BOOK_NOT_FOUND) await selectPlanet(planetId.value)
        else await loadTree()
    } finally {
        writing.value = false
    }
}
/** @param {import('@/api/catalogue').ReorderRequest} payload 抽屉冻结的快照与目标分组. */
function saveOrder(payload) {
    if (!orderOpen.value || payload.catalogue.parent_id !== folderId.value) return
    return writeChange(() => saveCatalogueOrder(payload), true)
}
/**
 * @param {CatalogueNode} node 目录或文件.
 * @param {boolean} visible 开关请求的目标状态.
 */
function changeDisplay(node, visible) {
    if (node.sort === null) return
    const status = visible === true ? 'visible' : 'hidden'
    return writeChange(() => setCatalogueDisplay(node.id, status), false, visible === true ? node : undefined)
}

onMounted(loadPlanets)
onBeforeUnmount(function () {
    active = false
    ++readVersion
})
</script>

3. 配置 Vite、端口与路径别名

在工程根目录创建 vite.config.js，配置 Vue 插件、局域网监听地址、开发服务端口 5062 和旧代码广泛使用的 @ 别名：

vite.config.js
import { fileURLToPath, URL } from 'node:url'
import { defineConfig } from 'vite'
import vue from '@vitejs/plugin-vue'

export default defineConfig({
  plugins: [vue()],
  server: {
    host: '0.0.0.0',
    port: 5062,
    strictPort: true,
  },
  resolve: {
    alias: {
      '@': fileURLToPath(new URL('./src', import.meta.url)),
    },
  },
})

server.host: '0.0.0.0' 让开发服务监听所有 IPv4 网卡地址，支持通过局域网 IP 访问。 server.port: 5062 指定 npm run dev 使用 5062 端口。 strictPort: true 表示端口被占用时直接报错，避免自动切换到其他端口。 修改配置后，重新运行 npm run dev。

参考：Vite 监听地址配置。

参考：Vite 开发服务端口配置。

npm run preview 使用独立的 preview.port 配置，默认端口为 4173； 本节的 server.port 用于开发服务。

定义用户 Store

用一个独立的用户 Store 管理 Token 和用户名，并提供登录成功、退出登录时调用的 actions：

src/stores/user.js
import { defineStore } from 'pinia'

/**
 * 用户状态。
 * @typedef {Object} UserState
 * @property {string} token 当前登录凭证；未登录时为空字符串。
 * @property {string} username 当前用户名；未登录时为空字符串。
 */

/**
 * 用户 Store 公开的会话操作，供 IDE 直接解析方法签名。
 * @typedef {Object} UserStoreActions
 * @property {function(string, string): void} setSession 更新会话，依次传入 Token 和用户名。
 * @property {function(): void} clearSession 清空用户会话。
 */

/**
 * 用户 Store 实例类型，包含 Pinia 标准能力和显式声明的业务成员。
 * @typedef {import('pinia').Store<'user', UserState, typeof userStoreOptions.getters, typeof userStoreOptions.actions> & UserStoreActions} UserStore
 */

/**
 * Token 在本地存储中的键名。
 * @type {string}
 */
const TOKEN_KEY = 'solar-notes-admin-token'

/**
 * 用户名在本地存储中的键名。
 * @type {string}
 */
const USERNAME_KEY = 'solar-notes-admin-username'

/**
 * 用户 Store 的配置对象，作为 defineStore 的第二个实参。
 * 包含初始状态、计算属性和会话操作。
 * @constant
 */
const userStoreOptions = {
  /**
   * 从浏览器本地存储恢复用户状态。
   * @returns {UserState} 包含 Token 和用户名的初始状态。
   */
  state: function () {
    let token = localStorage.getItem(TOKEN_KEY)
    let username = localStorage.getItem(USERNAME_KEY)

    if (token === null || token === undefined) {
      token = ''
    }

    if (username === null || username === undefined) {
      username = ''
    }

    return {
      token: token,
      username: username,
    }
  },
  getters: {
    /**
     * 判断当前状态中是否保存了 Token。
     * @param {UserState} state 用户状态。
     * @returns {boolean} 存在非空 Token 时为 true。
     */
    isLoggedIn: function (state) {
      return state.token !== ''
    },
  },
  actions: {
    /**
     * 更新用户会话，并同步保存到本地存储。
     * @this {UserState}
     * @param {string} token 登录凭证，不能为空字符串。
     * @param {string} username 用户名，不能为空字符串。
     * @returns {void}
     * @throws {TypeError} 任一参数不是非空字符串时抛出。
     */
    setSession: function (token, username) {
      if (typeof token !== 'string' || token === '') {
        throw new TypeError('token 必须是非空字符串')
      }

      if (typeof username !== 'string' || username === '') {
        throw new TypeError('username 必须是非空字符串')
      }

      this.token = token
      this.username = username
      localStorage.setItem(TOKEN_KEY, this.token)
      localStorage.setItem(USERNAME_KEY, this.username)
    },
    /**
     * 清空用户会话，并删除本地存储中的登录信息。
     * @this {UserState}
     * @returns {void}
     */
    clearSession: function () {
      this.token = ''
      this.username = ''
      localStorage.removeItem(TOKEN_KEY)
      localStorage.removeItem(USERNAME_KEY)
    },
  },
}

/**
 * 获取用户 Store 实例的函数；同一个 Pinia 实例内会复用该 Store。
 * 组件内调用 useUserStore()，组件外可调用 useUserStore(pinia)。
 * @type {import('pinia').StoreDefinition<'user', UserState, typeof userStoreOptions.getters, typeof userStoreOptions.actions>}
 */
const useUserStore = defineStore('user', userStoreOptions)

export default useUserStore

userStoreOptions 单独保存配置对象，再作为 defineStore() 的第二个实参。 JSDoc 使用 @type {类型} 标注常量类型，使用 @param 与 @returns 说明函数参数和返回值；@this 说明 actions 中访问的用户状态。

useUserStore 保存的是 defineStore() 返回的函数， 因此使用 @type 标注其 StoreDefinition 类型。 调用 useUserStore() 后取得的用户状态实例，类型才是 Store。 state、setSession 等直接定义的函数，继续使用 @param 和 @returns 描述参数与返回值。

用户 Store 模块先定义 useUserStore，再通过 export default useUserStore 默认导出。 页面和请求工具使用 import useUserStore from '@/stores/user' 导入。

页面和请求工具在 useUserStore() 调用表达式上使用 /** @type {import('@/stores/user').UserStore} */ (useUserStore()) 进行 JSDoc 类型断言。 UserStoreActions 显式列出会话操作的方法签名，UserStore 同时保留 Pinia 标准能力。 这种写法减少了 IDE 在调用处通过 ReturnType 和 Pinia 条件类型逐层解析业务成员的需要。 修改 Store 的公开成员时，同步维护这些声明。

类型断言将调用表达式直接标注为已声明的 UserStore， 用于兼容 IDE 对 Pinia 复杂返回类型的解析；方法参数和返回值仍按显式签名检查。

setSession(token, username) 的两个参数都必须是非空字符串。 函数会先完成两个参数的校验，再修改用户状态和本地存储；参数缺失、为 null、类型错误或为空字符串时， 抛出 TypeError。清空会话由 clearSession() 负责。

Pinia 本身不会自动把状态保存到浏览器。本示例沿用旧工程的 localStorage 方式， 在初始化时恢复会话，在 setSession 与 clearSession 中同步保存和清除。 首次访问时，本地存储中可能没有会话记录，因此 state 工厂中仍保留空值处理。 本方案面向浏览器端单页应用。

后续取得真实登录接口返回的 Token 和用户名后，依次将它们传给 setSession(token, username)。 以下函数可补充到登录页面的 script setup 中，并在登录请求成功后调用；字段按实际接口映射：

import useUserStore from '@/stores/user'

/**
 * 登录页面使用的用户 Store 实例。

const userStore = /** @type {import('@/stores/user').UserStore} */ (useUserStore())

function onLoginSuccess(token, username) {

const routes = [

在 Axios 拦截器中读取 Pinia 状态

Axios 拦截器位于组件外部，导入 stores/index.js 中创建的同一个实例， 在请求回调内通过 useUserStore(pinia) 获取用户 Store。 这样每次请求都会读取当前 Token，也不依赖组件上下文的自动注入。

src/utils/request.js
import axios from 'axios'
import pinia from '@/stores'
import useUserStore from '@/stores/user'

/**
 * 统一配置接口基础地址、超时和拦截器的 Axios 实例。
 * @type {import('axios').AxiosInstance}
 */
const request = axios.create({
  baseURL: import.meta.env.VITE_API_BASE_URL,
  timeout: 5000,
})

request.interceptors.request.use(
  /**
   * 读取最新的用户 Token，并添加到请求头。
   * @param {import('axios').InternalAxiosRequestConfig} config 请求配置。
   * @returns {import('axios').InternalAxiosRequestConfig} 已附加登录凭证的请求配置。
   */
  (config) => {
    /**
     * 与应用入口共享同一个 Pinia 实例的用户 Store。
     */
    const userStore = /** @type {import('@/stores/user').UserStore} */ (useUserStore(pinia))

    if (userStore.token !== '') {
      config.headers.Authorization = `Bearer ${userStore.token}`
    }

    return config
  }
)

request.interceptors.response.use(
  /**
   * 按业务响应码解包接口结果，失败时返回被拒绝的 Promise。
   * @param {import('axios').AxiosResponse} response Axios 响应对象。
   * @returns {*} 成功时返回业务数据，失败时返回拒绝的 Promise。
   */
  (response) => {
    /**
     * 后端返回的业务响应体。
     * @type {{code: number, data: *}}
     */
    const payload = response.data

    if (payload.code === 200) {
      return payload.data
    }

    return Promise.reject(payload)
  }
)

export default request

响应拦截器沿用旧工程的 { code, data } 协议，后端返回结构不同时按实际接口调整。 Store 的会话保存逻辑和 Axios 请求封装各自独立，无须互相导入。

参考：在组件外使用 Pinia Store。

npm run build

新工程沿用旧工程的路由、接口和请求工具分层，状态管理统一使用 stores/ 下的 Pinia Store。 入口、组件与 Axios 拦截器均按上述示例接入同一个 Pinia 实例。

在局域网内访问

按上面的配置设置 server.host: '0.0.0.0'，并重新运行 npm run dev。 本次检查到开发机的 en0 网卡地址为 192.168.1.131，访问方式如下：

访问位置浏览器地址
运行工程的这台 Machttp://localhost:5062/
同一局域网中可互通的其他机器http://192.168.1.131:5062/

0.0.0.0 是监听地址；浏览器中填写开发机实际的局域网 IP。 局域网 IP 可能随网络变化，请以当前网卡地址或 Vite 启动输出中的 Network 地址为准。 macOS 可运行以下命令，查看 inet 所在行的 IPv4 地址：

终端 · 查看 en0 网卡地址
ifconfig en0

直接通过 IP 访问时，Vite 默认的 allowedHosts 配置已允许 IP 地址，使用默认值即可。

参考：Vite 主机名校验配置。

其他机器无法打开页面时

  • 确认两台机器的网络可以互通；访客 Wi-Fi 或路由器的客户端隔离可能阻止设备之间连接。
  • 确认开发服务仍在运行，并且 Vite 启动输出中的端口为 5062。
  • 如果 macOS 防火墙阻止连接，在“系统设置 → 网络 → 防火墙 → 选项”中允许运行 Vite 的 Node.js 进程接收传入连接。

参考：Apple 防火墙访问设置。

页面打开后，接口地址也需要可达

浏览器中的 localhost 指访问者自己的机器。 如果 VITE_API_BASE_URL 仍为 http://localhost:8080， 其他机器会向自身的 8080 端口发送请求。

直接访问后端时，将该变量设为其他机器可达的后端地址；例如后端也运行在这台 Mac 的 8080 端口， 可使用 http://192.168.1.131:8080，并让后端允许局域网连接及对应前端来源的跨域请求。 另一种方式是使用 Vite 的 server.proxy 转发请求，前端统一使用相对路径访问接口。

参考：Vite 开发代理配置。

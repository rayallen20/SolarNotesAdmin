天体管理页面重构与 API 联调指南

打开天体管理页即可看到太阳与八颗行星是否齐全。管理员可以补建缺项，也可以直接从列表编辑现有天体的标题和简介。

1. 页面与修改范围

沿用 /admin/bodies 和现有侧边栏。在列表上方增加完整性检测卡，在每行末尾增加“编辑”。创建和编辑共用右侧抽屉，表头和单元格沿用工程的居中样式。

修改顺序文件要改的行为
1src/constants/apiCode.js让天体编辑识别记录消失和编辑冲突，兼容现有书籍、目录调用方。
2src/api/body.js扩展状态读取，复用列表行解析，新增更新方法。
3src/views/BodyListPage.vue将原有列表计数升级为独立检测卡，把仅支持创建的抽屉扩展为创建和编辑两种模式。

按第 3 至 5 节顺序应用三个完整文件示例。原有 getBodyList() 仍供书籍和目录页面使用，getBodyStatus() 的返回类型随天体页面一起更新。页面继续使用既有路由、Axios 实例、Pinia 登录态和 Element Plus。

2. 请求与响应

以下路径与 Go 路由一致。开发环境下浏览器会在路径前加 /api，公共请求实例附加 Authorization: Bearer <access_token>。在浏览器 Network 中看完整的 {code, message, data}，在请求模块中直接读拦截器解包后的 data。HTTP 200 也可能携带失败业务码。

页面动作方法和路径请求与成功结果
进入页面或重试检测GET /v1/planet/status无请求体。data.planets 给出九项明细，data.readiness 给出是否齐全、总数、已创建数和缺失标识。
加载或刷新列表GET /v1/planet/list无请求体。data.planets 是已创建记录数组，空表返回 []。每行有 id, name, title, intro。
确认创建POST /v1/planet/createJSON 提交 planet.{name,title,intro}，成功为 data: {}。名称取自缺失清单，ID 由后端生成。
保存编辑POST /v1/planet/updateJSON 提交 planet.{id,expected,target}，成功返回一条 data.planet，结构与列表行相同。

4. 扩展天体请求模块

创建仍调用 addBody(form)，选择所属天体的其他页面仍调用 getBodyList()。本页通过以下四个函数读取或提交数据。

函数实现要点
getBodyStatus()返回 {planets, readiness}。校验九个名称及布尔状态，同时核对两个计数、缺失清单、ready 与明细。旧服务若只返回 planets，让检测卡进入失败态，以便发现部署版本不一致。
getBodyList()保持 BodyRecord[] 契约与接口顺序。使用原始字符串读取标题和简介，保留历史空串及首尾空白，供编辑抽屉建立准确快照。
addBody(form)只发送名称、标题、简介。修剪输入后提交，成功即由页面重读列表及完整性。
updateBody(id, expected, target)保持 ID 为字符串以保留雪花 ID 精度。原样提交两个旧值，只修剪目标值。检查响应行及其 ID，按同一套规则读取列表和更新结果。
完整文件src/api/body.js
import request from '@/utils/request'
import {bodyOptions} from '@/constants/body'
import {isRecord} from '@/utils/typeGuard'
import {readRequiredStringField, readSafeIntegerField, readStringField} from '@/utils/fileReader'

/**
 * @typedef {Object} BodyStatus 固定天体的创建状态.
 * @property {string} name 固定英文标识.
 * @property {boolean} exist 已创建时为 true.
 */

/**
 * @typedef {Object} BodyReadiness 后端完整性检测结果.
 * @property {boolean} ready 九个必需天体均已创建时为 true.
 * @property {number} required_count 必需天体总数.
 * @property {number} created_count 已创建的必需天体数.
 * @property {string[]} missing_names 按固定顺序列出的缺失标识.
 */

/**
 * @typedef {Object} BodyStatusResult 同一次检测的明细和摘要.
 * @property {BodyStatus[]} planets 太阳与八颗行星的创建状态.
 * @property {BodyReadiness} readiness 完整性摘要.
 */

/**
 * @typedef {Object} BodyFields 可编辑的字段.
 * @property {string} title 天体标题.
 * @property {string} intro 天体简介.
 */

/**
 * @typedef {Object} BodyRecord 数据库中已创建的天体.
 * @property {string} id 十进制字符串 ID.
 * @property {string} name 大小写敏感的原始标识.
 * @property {string} title 天体标题原文.
 * @property {string} intro 天体简介原文.
 */

/**
 * @typedef {Object} AddBodyForm 创建表单.
 * @property {string} name 待创建天体的固定标识.
 * @property {string} title 天体标题.
 * @property {string} intro 天体简介.
 */

/**
 * 读取已经由 Axios 解包的数据中的 planets.
 * @param {unknown} data 业务响应中的 data.
 * @returns {unknown[]} 待校验的数组.
 */
function readPlanets(data) {
    if (isRecord(data) === false || Array.isArray(data.planets) === false) {
        throw new Error('天体接口响应缺少 planets 数组')
    }
    return data.planets
}

/**
 * 逐项核对后端约定的九个固定标识及其顺序.
 * @param {unknown[]} items 状态明细原值.
 * @returns {BodyStatus[]} 可供表单使用的创建状态.
 */
function readStatuses(items) {
    if (items.length !== bodyOptions.length) {
        throw new Error('天体状态明细不完整')
    }
    /** @type {BodyStatus[]} */
    const planets = []
    for (const [index, item] of items.entries()) {
        if (isRecord(item) === false) {
            throw new Error('天体状态记录格式异常')
        }
        const name = readRequiredStringField(item, 'name')
        if (name !== bodyOptions[index].name || typeof item.exist !== 'boolean') {
            throw new Error('天体状态的标识, 顺序或创建标记异常')
        }
        planets.push({name, exist: item.exist})
    }
    return planets
}

/**
 * 核对同一响应的摘要与逐项状态, 拒绝相互矛盾的结果.
 * @param {unknown} value 后端返回的 readiness.
 * @param {BodyStatus[]} planets 已校验的明细.
 * @returns {BodyReadiness} 可用于提示和计数的摘要.
 */
function readReadiness(value, planets) {
    if (isRecord(value) === false || typeof value.ready !== 'boolean' || Array.isArray(value.missing_names) === false) {
        throw new Error('天体完整性摘要格式异常')
    }

    const required = readSafeIntegerField(value, 'required_count', planets.length, planets.length)
    const created = readSafeIntegerField(value, 'created_count', 0, required)
    /** @type {string[]} */
    const missing = []
    for (const planet of planets) {
        if (planet.exist === false) {
            if (value.missing_names[missing.length] !== planet.name) {
                throw new Error('缺失天体与逐项状态不一致')
            }
            missing.push(planet.name)
        }
    }
    if (value.missing_names.length !== missing.length ||
        created !== required - missing.length || value.ready !== (missing.length === 0)) {
        throw new Error('天体完整性摘要与逐项状态不一致')
    }
    return {ready: value.ready, required_count: required, created_count: created, missing_names: missing}
}

/**
 * 同时获取固定天体的创建状态和完整性检测结论.
 * @returns {Promise<BodyStatusResult>} 已校验的一次检测结果.
 */
async function getBodyStatus() {
    /** @type {unknown} */
    const data = await request.get('/v1/planet/status')
    if (isRecord(data) === false) {
        throw new Error('天体状态响应格式异常')
    }
    const planets = readStatuses(readPlanets(data))
    return {planets, readiness: readReadiness(data.readiness, planets)}
}

/**
 * 从列表元素或编辑响应读取完整天体, 保留历史标题和简介原文.
 * @param {unknown} value 一条天体记录.
 * @returns {BodyRecord} 已校验的天体.
 */
function readBodyRecord(value) {
    if (isRecord(value) === false) {
        throw new Error('天体记录格式异常')
    }
    return {
        id: readRequiredStringField(value, 'id'),
        name: readRequiredStringField(value, 'name'),
        title: readStringField(value, 'title'),
        intro: readStringField(value, 'intro'),
    }
}

/**
 * 按服务端顺序读取全部已创建记录, 包括历史名称.
 * @returns {Promise<BodyRecord[]>} 列表为空时返回空数组.
 */
async function getBodyList() {
    const data = await request.get('/v1/planet/list')
    return readPlanets(data).map(readBodyRecord)
}

/**
 * 创建一个缺失天体.
 * @param {AddBodyForm} form 已校验的创建表单.
 * @returns {Promise<void>} 创建成功后由页面重查列表与状态.
 */
async function addBody(form) {
    const data = await request.post('/v1/planet/create', {
        planet: {name: form.name, title: form.title.trim(), intro: form.intro.trim()},
    })
    if (isRecord(data) === false) {
        throw new Error('创建响应格式异常, 请刷新列表确认结果')
    }
}

/**
 * 按 ID 保存标题和简介.
 * @param {string} id 列表中的字符串 ID.
 * @param {BodyFields} expected 打开抽屉或确认最新版本时取得的原文.
 * @param {BodyFields} target 本次表单的目标值.
 * @returns {Promise<BodyRecord>} 数据库已保存的完整行.
 */
async function updateBody(id, expected, target) {
    const data = await request.post('/v1/planet/update', {
        planet: {
            id,
            expected: {title: expected.title, intro: expected.intro},
            target: {title: target.title.trim(), intro: target.intro.trim()},
        },
    })
    if (isRecord(data) === false) {
        throw new Error('编辑响应格式异常, 请刷新列表确认结果')
    }
    const saved = readBodyRecord(data.planet)
    if (saved.id !== id) {
        throw new Error('返回的天体 ID 与请求不一致, 请刷新列表确认结果')
    }
    return saved
}

export {getBodyStatus, getBodyList, addBody, updateBody}

5. 实现检测卡、列表和表单抽屉

进页时同时读取列表和完整性

分别保存列表的加载结果和检测卡的状态。两次读取并行发出，任意一项失败，另一项仍可完成。检测卡的“重新检测”只读取 status。若短时间触发多次读取，仅接受最新一轮同类请求的结果。

检测卡状态显示内容可用操作
检测中正在检测，计数显示占位。等待本次检测结果。已加载的行仍可编辑。
有缺项尚有未被创建的天体，前台页面无法正常启动。显示后端计数，用 getBodyLabel() 将缺失标识转为中文。创建抽屉只提供缺项。只剩一个时自动选中。
九项齐全太阳与八颗行星均已创建，计数为 9 / 9。禁用创建入口，编辑仍可用。
检测失败暂时无法确认天体是否齐全，保留重新检测入口。已加载的行仍可编辑。创建必须取得有效的缺失清单才能提交。

把身份、原文和草稿分开保存

点击“编辑”时，使用列表行回填抽屉。中文名称和 name 只读，id 仅用于定位。将 title 与 intro 复制到编辑前的快照和表单草稿中。这样输入或取消都不会提前改变列表。

提交时以修剪首尾空白后的文本校验。标题为 1–20 个 Unicode 码点，简介为 1–150 个，可用 Array.from(text).length 计数。诸如 🪐 的字符按 1 个码点计。写请求进行中锁住提交和关闭入口。

                    <p v-else-if="allCreated">太阳与八颗行星均已创建。</p>
                </template>
                <el-form-item v-else label="天体">
                    <span class="identity">{{ getBodyLabel(original.name) }} <code>{{ original.name }}</code></span>
                </el-form-item>
                <el-form-item label="天体标题" :error="titleError">
                    <el-input v-model="form.title" placeholder="请输入天体标题" @input="titleError = ''" />
                    <small>{{ lengthOf(form.title) }} / {{ BODY_TITLE_MAX_LENGTH }}</small>
                </el-form-item>
                <el-form-item label="天体简介" :error="introError">
                    <el-input v-model="form.intro" type="textarea" :rows="6" placeholder="请输入天体简介" @input="introError = ''" />
                    <small>{{ lengthOf(form.intro) }} / {{ BODY_INTRO_MAX_LENGTH }}</small>
                </el-form-item>
                <div v-if="review" class="review" role="alert">
                    <strong>{{ notice }}</strong>
                    <p v-if="reading">正在读取服务器原文…</p>
                    <p v-else-if="listError !== ''">{{ listError }}
                        <el-button link type="primary" :disabled="busy" @click="recoverEdit">重新读取原文</el-button>
                    </p>
                    <template v-else-if="latest !== null">
                        <p>服务器标题：<span>{{ latest.title }}</span></p>
                        <p>服务器简介：<span>{{ latest.intro }}</span></p>
                        <p>请与上方草稿核对，确认后再保存。</p>
                    </template>
                </div>
                <p v-else-if="notice !== ''" role="alert">{{ notice }}</p>
            </el-form>
            <template #footer>
                <el-button :disabled="busy" @click="drawer = false">取消</el-button>
                <el-button type="primary" :loading="writing" :disabled="!canSave" @click="save">
                    {{ original === null ? '确认创建' : (review ? '确认并保存修改' : '保存修改') }}
                </el-button>
            </template>
        </el-drawer>
    </section>
</template>

<script setup>
import {computed, onBeforeUnmount, onMounted, reactive, ref} from 'vue'
import {ElMessage} from 'element-plus'
import {ArrowRight, Plus} from '@element-plus/icons-vue'
import AdminFilterSelect from '@/components/AdminFilterSelect.vue'
import BodyIcon from '@/components/BodyIcon.vue'
import {addBody, getBodyList, getBodyStatus, updateBody} from '@/api/body'
import {BODY_TITLE_MAX_LENGTH, BODY_INTRO_MAX_LENGTH} from '@/constants/body'
import {API_CODE} from '@/constants/apiCode'
import {getBodyLabel} from '@/utils/body'
import {getApiErrorCode, getApiErrorMessage, isAuthError} from '@/utils/apiError'

defineOptions({name: 'BodyListPage'})

/** @typedef {import('@/api/body').BodyRecord} BodyRecord */
/** @typedef {import('@/api/body').BodyReadiness} BodyReadiness */
/** @type {import('vue').Ref<BodyRecord[]>} 已创建的记录, 保留服务端顺序. */
const rows = ref([])
/** @type {import('vue').Ref<BodyReadiness|null>} 已核验的后端检测摘要. */
const readiness = ref(null)
/** 列表正在加载. */
const reading = ref(false)
/** 首屏和复检期间的独立状态. */
const checking = ref(true)
/** 列表读取失败说明. */
const listError = ref('')
/** 完整性检测失败说明. */
const checkError = ref('')
/** 创建或编辑抽屉是否可见. */
const drawer = ref(false)
/** 是否已有写请求在途. */
const writing = ref(false)
/** @type {import('vue').Ref<BodyRecord|null>} 打开编辑抽屉时的原始行. */
const original = ref(null)
/** @type {import('vue').Ref<BodyRecord|null>} 冲突后读回的最新服务器原文. */
const latest = ref(null)
/** 是否需要对照服务器原文后再次确认. */
const review = ref(false)
/** 当前草稿, 独立于列表和编辑快照. */
const form = reactive({name: '', title: '', intro: ''})
/** 天体标题的校验说明. */
const titleError = ref('')
/** 天体简介的校验说明. */
const introError = ref('')
/** 需要管理员处理的业务提示. */
const notice = ref('')
/** 组件仍接收请求结果. */
let active = true
/** 列表请求序号. */
let listVersion = 0
/** 检测请求序号. */
let statusVersion = 0

/** 服务端确认九项都存在. */
const allCreated = computed(function () { return readiness.value !== null && readiness.value.ready === true })
/** @type {import('vue').ComputedRef<string[]>} 按后端顺序列出的可创建标识. */
const missingNames = computed(function () {
    if (readiness.value === null) return []
    return readiness.value.missing_names
})
/** 核对原文或提交期间锁定抽屉. */
const busy = computed(function () { return writing.value || (review.value && reading.value) })
/** 仅允许使用可信的创建选项或已经取得的编辑快照. */
const canSave = computed(function () {
    if (busy.value) return false
    if (original.value !== null) return review.value === false || latest.value !== null
    return missingNames.value.includes(form.name)
})

/**
 * @param {string} text 待计数的正文.
 * @returns {number} Unicode 码点数.
 */
function lengthOf(text) {
    return Array.from(text).length
}

/**
 * 读取列表, 用序号丢弃已过期请求.
 * @returns {Promise<void>}
 */
async function loadList() {
    const version = ++listVersion
    reading.value = true
    listError.value = ''
    try {
        const result = await getBodyList()
        if (active && version === listVersion) rows.value = result
    } catch (error) {
        if (active && version === listVersion && isAuthError(error) === false) {
            listError.value = getApiErrorMessage(error, '天体列表加载失败')
        }
    } finally {
        if (active && version === listVersion) reading.value = false
    }
}

/**
 * 读取服务端摘要. 检测期间及失败后均显示未知状态.
 * @returns {Promise<void>}
 */
async function loadStatus() {
    const version = ++statusVersion
    checking.value = true
    readiness.value = null
    checkError.value = ''
    try {
        const result = await getBodyStatus()
        if (active === false || version !== statusVersion) return
        readiness.value = result.readiness
        if (drawer.value && original.value === null) {
            if (form.name !== '' && missingNames.value.includes(form.name) === false) form.name = ''
            if (form.name === '' && notice.value === '' && missingNames.value.length === 1) {
                form.name = missingNames.value[0]
            }
        }
    } catch (error) {
        if (active && version === statusVersion && isAuthError(error) === false) {
            checkError.value = getApiErrorMessage(error, '检测失败, 请重试')
        }
    } finally {
        if (active && version === statusVersion) checking.value = false
    }
}

/**
 * 首次进入及创建后并行读取, 两类失败可分别重试.
 * @returns {Promise<void[]>}
 */
function refreshBoth() {
    return Promise.all([loadList(), loadStatus()])
}

/** @param {BodyRecord|null} row 编辑时的列表行, 创建时为 null. */
function openForm(row) {
    original.value = row
    review.value = false
    latest.value = null
    notice.value = ''
    titleError.value = ''
    introError.value = ''
    form.name = ''
    form.title = ''
    form.intro = ''
    if (row !== null) {
        original.value = {...row}
        form.title = row.title
        form.intro = row.intro
    } else if (missingNames.value.length === 1) {
        form.name = missingNames.value[0]
    }
    drawer.value = true
}

/** 打开创建抽屉. 无可信摘要时可以在抽屉中重试检测. */
function openCreate() {
    if (allCreated.value === false && writing.value === false) openForm(null)
}

/** @param {BodyRecord} row 已创建天体的列表行. */
function openEdit(row) {
    if (writing.value === false && reading.value === false) openForm(row)
}

/**
 * 记录失效后关闭编辑入口并同步两组读数据.
 * @returns {Promise<void>}
 */
async function recoverMissing() {
    drawer.value = false
    ElMessage.warning('该天体已不存在, 已重新读取列表和创建状态')
    await refreshBoth()
}

/**
 * 重新取得编辑原文, 供管理员与仍在抽屉中的草稿比较.
 * @returns {Promise<void>}
 */
async function recoverEdit() {
    review.value = true
    latest.value = null
    const row = original.value
    if (row === null) return
    await loadList()
    if (active === false || listError.value !== '' || reading.value) return
    const found = rows.value.find(function (item) { return item.id === row.id })
    if (found === undefined) {
        await recoverMissing()
        return
    }
    latest.value = {...found}
}

/**
 * @param {unknown} error 创建或编辑的异常.
 * @returns {Promise<void>}
 */
async function handleWriteError(error) {
    if (active === false || isAuthError(error)) return
    const code = getApiErrorCode(error)
    if (original.value !== null && code === API_CODE.BODY_NOT_FOUND) {
        await recoverMissing()
        return
    }
    if (original.value !== null && (code === API_CODE.BODY_EDIT_CHANGED || code === undefined)) {
        notice.value = code === API_CODE.BODY_EDIT_CHANGED ? '天体已被其他管理员修改' : '保存结果待确认'
        await recoverEdit()
        return
    }
    if (original.value === null && (code === API_CODE.BODY_ALREADY_EXISTS || code === undefined)) {
        notice.value = code === API_CODE.BODY_ALREADY_EXISTS ? '该天体已被创建, 请重新核对' : '创建结果待确认, 请核对最新列表'
        await refreshBoth()
        return
    }
    notice.value = getApiErrorMessage(error, '保存失败, 请稍后重试')
}

/**
 * 校验目标, 创建后复检, 编辑成功按字符串 ID 替换单行.
 * @returns {Promise<void>}
 */
async function save() {
    if (drawer.value === false || canSave.value === false) return
    const title = form.title.trim()
    const intro = form.intro.trim()
    titleError.value = ''
    introError.value = ''
    if (title === '' || lengthOf(title) > BODY_TITLE_MAX_LENGTH) {
        titleError.value = '天体标题需要 1 至 ' + BODY_TITLE_MAX_LENGTH + ' 个字符'
    }
    if (intro === '' || lengthOf(intro) > BODY_INTRO_MAX_LENGTH) {
        introError.value = '天体简介需要 1 至 ' + BODY_INTRO_MAX_LENGTH + ' 个字符'
    }
    if (titleError.value !== '' || introError.value !== '') return
    writing.value = true
    try {
        if (original.value === null) {
            await addBody({name: form.name, title, intro})
            if (active === false) return
            drawer.value = false
            ElMessage.success('天体已创建')
            await refreshBoth()
        } else {
            let before = original.value
            if (review.value && latest.value !== null) before = latest.value
            const expected = {title: before.title, intro: before.intro}
            const updated = await updateBody(before.id, expected, {title, intro})
            if (active === false) return
            rows.value = rows.value.map(function (row) {
                if (row.id === updated.id) return updated
                return row
            })
            drawer.value = false
            ElMessage.success('天体信息已更新')
        }
    } catch (error) {
        await handleWriteError(error)
    } finally {
        if (active) writing.value = false
    }
}

onMounted(refreshBoth)
onBeforeUnmount(function () { active = false })
</script>

测试步骤Network 与页面的预期结果
① 打开只有 7 个固定天体的测试数据GET list 和 GET status 各一次。两项若缺 Uranus 与 Neptune，页面显示 7 / 9、天王星、海王星和缺项警告。即使另有历史记录使列表总数达到 9，也应显示 7 / 9。
② 单独让 status 失败，再恢复可以在浏览器中只屏蔽 status 后刷新。列表正常、检测显示未知，仍可编辑。解除屏蔽并点“重试检测”，Network 应只有一次新的 status。反向只屏蔽 list 时，检测结果仍独立显示。
③ 在测试数据上补齐最后两个天体每次确认都只发一个 POST create，成功后各读一次 list 和 status。看到 8 / 9、9 / 9，最后创建入口禁用。两页同时创建同一天体时，后提交的页面处理 20101。
④ 取消编辑，再打开并保存取消时无 POST，表格保留原文。保存时核对 expected 与打开抽屉时一致，target 是修剪后的两项文本。成功回显服务器返回行，未触发完整性复检。
⑤ 用两个标签页修改同一天体按第 6 节复现 20103，乙的草稿保留，只有明确确认后才发第二次 update。可同时验证保留服务器最新标题和自己修改的简介。
⑥ 检查边界与记录消失空白标题、21 个标题码点、151 个简介码点应阻止提交；20 个 🪐 标题、150 个简介码点可提交。用测试数据或响应模拟让 update 返回 20102，应关闭抽屉并刷新两项数据。
⑦ 回归书籍与目录管理分别进入两个页面，仍可加载已创建的天体选项。两处共用 getBodyList() 和原有错误码名称。

取得一个已确认的创建成功响应后，可临时阻断后续 status。列表应显示新记录，检测卡可单独重试，Network 中不会自动重复提交 create。

书籍管理页面重构指南

把 /admin/books 从全库分页改为按天体管理完整书单。管理员先选天体，再添加书籍、编辑标题和显隐，或一次保存整份阅读顺序。

1. 核对已经实现的后端接口

以 SolarNotesAdminAPI 的提交 34222e4 为核对基准。下列 5 个接口均已注册，控制器、请求结构和持久化逻辑均已实现。表中的 data 指 HTTP 响应的有效载荷，Axios 拦截器已将其解包。路由与鉴权 · 书籍控制器 · 数据库读写

操作已实现的路由请求与成功结果
进入页面GET /v1/planet/list无请求体。data.planets 含已创建的天体，每项有 id, name, title, intro。选项标签复用 getBodyLabel(name)。
选天体或刷新POST /v1/book/options提交 planet.id。data.books 返回该天体下显示和隐藏的全部书籍，按 sort, id 升序；空书单为 []。
确认添加POST /v1/book/create只提交 book.planet.id、book.title、book.display_status。成功为 data: {}，再查一次 options 获取实际 ID 和序号。
保存编辑POST /v1/book/update提交 book.id、book.expected 与 book.target。成功为 data.book，含 id, title, sort, display_status。
保存书序POST /v1/book/reorder提交 planet.id、planet.expected.books 与 planet.target.books。两个数组均由 {id} 组成。data.books 返回提交时的完整书单，按新顺序排列。

这五个接口都使用管理员 Bearer token。业务层成功为数值型 code: 200，业务失败也可能返回 HTTP 200。复用 src/utils/request.js 发请求，业务方法只写 /v1/…。本项目开发时由 /api 代理转发。天体和书籍 ID 都保留为十进制字符串。

创建时以实际后端为准。 当前请求类型已移除 sort，现用 JSON 绑定会忽略多传的该字段。前期设计中“用旧序号核对预期末位”的过渡逻辑没有落地。后端锁定天体后统计全部书籍，确认已有序号连续，再分配 N + 1。历史序号有 0、跳号或序号空间耗尽时，创建回滚并返回业务码 500；当前后端已经移除旧排序占用码 20201。创建请求 · 错误映射

4. 整理书籍 API 与目录页共用的解析器

目前 getBookOptions() 在目录 API 模块中，旧 getBookList() 的分页记录还要求一个嵌套的 planet 对象。将按天体书单和书籍摘要放到 book.js 后，书籍列表、编辑结果、调序结果和目录树的书籍部分就能使用同一套字段定义。

新页面调用页面拿到什么
getBodyList()已存在于 src/api/body.js，返回天体数组。
getBookOptions(planetId)返回 BookOption[]，其中标题保留后端原文，排序读取兼容历史值 0。
addBook(form)接受 planet_id, title, display_status。成功后由页面重读书单。
updateBook(bookId, expected, target)返回已保存的一本书。快照里的 title 原样传回，包括历史首尾空白或空字符串。目标标题去除首尾空白。
reorderBooks(planetId, expectedIds, targetIds)API 层把字符串 ID 包装成 {id}，成功后返回最新书单，可直接替换页面列表。
完整文件src/api/book.js
import request from '@/utils/request'
import {isRecord} from '@/utils/typeGuard'
import {readRequiredStringField, readSafeIntegerField, readStringField} from '@/utils/fileReader'
import {BOOK_SORT_MAX} from '@/constants/book'
import {readPagination, validatePageItemCount, validatePaginationInput} from '@/types/pagination'

/** @typedef {import('@/constants/book').BookDisplayStatus} DisplayStatus */
/**
 * 编辑面板的快照或目标字段.
 * @typedef {Object} BookFields
 * @property {string} title 书名.
 * @property {DisplayStatus} display_status 书籍自身的显示状态.
 */
/** @typedef {BookFields & {id: string, sort: number}} BookOption 书单与目录树共用的书籍摘要. */
/** @typedef {BookOption & {planet: import('@/api/body').BodyRecord}} BookRecord 旧分页接口的一行. */
/** @typedef {{books: BookRecord[], pagination: import('@/types/pagination').Pagination}} BookListResult */
/** @typedef {BookFields & {planet_id: string}} CreateBookInput 创建时由后端分配排序值. */

/**
 * 读取响应中的显隐枚举, 供书籍和目录共用.
 * @param {Record<string, unknown>} record 含 display_status 的记录.
 * @returns {DisplayStatus} 已检查的枚举.
 */
function readDisplayStatus(record) {
    const status = readRequiredStringField(record, 'display_status')
    if (status !== 'visible' && status !== 'hidden') {
        throw new Error('显示状态无效')
    }
    return status
}

/**
 * 读取一条书籍摘要. 保留历史书名原文及目录模块支持的序号 0.
 * @param {unknown} value 原始响应中的记录.
 * @returns {BookOption} 对应的书籍摘要.
 */
function readBookOption(value) {
    if (isRecord(value) === false) {
        throw new Error('书籍记录格式异常')
    }
    return {
        id: readRequiredStringField(value, 'id'),
        title: readStringField(value, 'title'),
        sort: readSafeIntegerField(value, 'sort', 0, BOOK_SORT_MAX),
        display_status: readDisplayStatus(value),
    }
}

/**
 * 按服务端顺序读取选项或重排响应的整份书单.
 * @param {unknown} data 拦截器解包后的数据.
 * @returns {BookOption[]} 当前天体的全部书籍.
 */
function readBookOptions(data) {
    if (isRecord(data) === false || Array.isArray(data.books) === false) {
        throw new Error('书籍响应缺少 books 数组')
    }
    return data.books.map(readBookOption)
}

/**
 * 读取既有全库分页接口的一条记录, 保持原来的正整数校验.
 * @param {unknown} value 含所属天体的完整记录.
 * @returns {BookRecord} 经过校验的分页行.
 */
function readBook(value) {
    if (isRecord(value) === false || isRecord(value.planet) === false) {
        throw new Error('书籍记录或所属天体格式异常')
    }
    const book = readBookOption(value)
    if (book.sort === 0) {
        throw new Error('接口响应中的 sort 超出允许范围')
    }
    return {
        ...book,
        planet: {
            id: readRequiredStringField(value.planet, 'id'),
            name: readRequiredStringField(value.planet, 'name'),
            title: readStringField(value.planet, 'title'),
            intro: readStringField(value.planet, 'intro'),
        },
    }
}

/**
 * 读取原有全库分页列表.
 * @param {number} currentPage 当前页码.
 * @param {number} itemPerPage 每页条数.
 * @returns {Promise<BookListResult>} 当前页及分页信息.
 */
async function getBookList(currentPage, itemPerPage) {
    validatePaginationInput(currentPage, itemPerPage)
    /** @type {unknown} */
    const data = await request.post('/v1/book/list', {pagination: {currentPage, itemPerPage}})
    if (isRecord(data) === false || Array.isArray(data.books) === false) {
        throw new Error('书籍列表响应缺少 books 数组')
    }
    const pagination = readPagination(data.pagination, currentPage, itemPerPage)
    validatePageItemCount(data.books.length, pagination)
    return {books: data.books.map(readBook), pagination}
}

/**
 * 读取当前天体的全部书籍, 包括隐藏书籍.
 * @param {string} planetId 天体 ID.
 * @returns {Promise<BookOption[]>} 按后端书序排列的摘要.
 */
async function getBookOptions(planetId) {
    const data = await request.post('/v1/book/options', {planet: {id: planetId}})
    return readBookOptions(data)
}

/**
 * 追加一本书, 排序值由服务端在事务内分配.
 * @param {CreateBookInput} form 已通过页面校验的表单.
 * @returns {Promise<void>} 提交成功, 调用方另行刷新书单.
 */
async function addBook(form) {
    const data = await request.post('/v1/book/create', {
        book: {
            planet: {id: form.planet_id},
            title: form.title.trim(),
            display_status: form.display_status,
        },
    })
    if (isRecord(data) === false) {
        throw new Error('创建响应格式异常, 请刷新书单确认结果')
    }
}

/**
 * 更新书名及书籍自身的显隐.
 * @param {string} bookId 书籍 ID.
 * @param {BookFields} expected 打开抽屉时的原始值, 包括书名首尾空白.
 * @param {BookFields} target 保存时的表单值.
 * @returns {Promise<BookOption>} 后端已保存的书籍摘要.
 */
async function updateBook(bookId, expected, target) {
    const data = await request.post('/v1/book/update', {
        book: {
            id: bookId,
            expected: {title: expected.title, display_status: expected.display_status},
            target: {title: target.title.trim(), display_status: target.display_status},
        },
    })
    if (isRecord(data) === false) {
        throw new Error('编辑响应缺少 book 对象, 请刷新书单确认结果')
    }
    const book = readBookOption(data.book)
    if (book.id !== bookId) {
        throw new Error('编辑响应中的书籍 ID 与请求不一致')
    }
    return book
}

/**
 * 将 ID 包装为书籍引用, 保留 ID 的字符串精度.
 * @param {string} id 书籍 ID.
 * @returns {{id: string}} 请求中的书籍引用.
 */
function bookRef(id) {
    return {id}
}

/**
 * 保存一个天体的完整书序, 以数组位置表达排序.
 * @param {string} planetId 天体 ID.
 * @param {string[]} expectedIds 开始调序时的全部 ID.
 * @param {string[]} targetIds 调整后的全部 ID.
 * @returns {Promise<BookOption[]>} 后端返回的最新书单.
 */
async function reorderBooks(planetId, expectedIds, targetIds) {
    const data = await request.post('/v1/book/reorder', {
        planet: {
            id: planetId,
            expected: {books: expectedIds.map(bookRef)},
            target: {books: targetIds.map(bookRef)},
        },
    })
    const books = readBookOptions(data)
    if (books.length !== targetIds.length || books.some(function (book, index) {
        return book.id !== targetIds[index]
    })) {
        throw new Error('保存书序响应与目标顺序不一致, 请刷新书单确认结果')
    }
    return books
}

export {getBookList, getBookOptions, addBook, updateBook, reorderBooks, readBookOption, readDisplayStatus}

import request from '@/utils/request'
import {isRecord} from '@/utils/typeGuard'
import {readRequiredStringField, readSafeIntegerField, readStringField} from '@/utils/fileReader'
import {getBookOptions, readBookOption, readDisplayStatus} from '@/api/book'

/** @typedef {import('@/constants/book').BookDisplayStatus} DisplayStatus */
/** @typedef {import('@/api/book').BookOption} BookOption */
/**
 * @typedef {Object} CatalogueNode
 * @property {string} id 目录节点 ID.
 * @property {string} parent_id 直接父节点的 ID, 根节点为 "0".
 * @property {string} name 目录名或文件名.
 * @property {'folder' | 'file'} type 节点类型.
 * @property {number | null} sort 序号, null 表示待排.
 * @property {DisplayStatus} display_status 节点自身的显隐.
 * @property {CatalogueNode[]} children 已按阅读顺序排列的直接子节点.
 */
/** @typedef {BookOption & {catalogues: CatalogueNode[]}} CatalogueBook */
/** @typedef {{sorted_children: {id: string}[], unsorted_children: {id: string}[]}} OrderGroups */
/** @typedef {{catalogue: {parent_id: string, expected: OrderGroups, target: OrderGroups}}} ReorderRequest */
/** @typedef {{id: string, title: string, content: string}} ArticlePreview */

/**
 * 读取后端已组装的节点, 保留各层 children 的顺序.
 * @param {unknown} value 目录树中的一个节点.
 * @returns {CatalogueNode} 可交给页面的节点.
 */
function readCatalogueNode(value) {
    if (isRecord(value) === false || Array.isArray(value.children) === false) {
        throw new Error('目录节点缺少 children 数组')
    }
    const type = readRequiredStringField(value, 'type')
    if (type !== 'folder' && type !== 'file') {
        throw new Error('目录节点类型无效')
    }
    const parentId = readRequiredStringField(value, 'parent_id')
    let sort = null
    if (parentId === '0') {
        sort = readSafeIntegerField(value, 'sort', 0, 0)
    } else if (value.sort !== null) {
        sort = readSafeIntegerField(value, 'sort', 1, Number.MAX_SAFE_INTEGER)
    }
    return {
        id: readRequiredStringField(value, 'id'),
        parent_id: parentId,
        name: readStringField(value, 'name'),
        type,
        sort,
        display_status: readDisplayStatus(value),
        children: value.children.map(readCatalogueNode),
    }
}

/**
 * 读取一本书和它的完整目录树.
 * @param {string} bookId 书籍 ID.
 * @returns {Promise<CatalogueBook>} 尚未导入目录时 catalogues 是空数组.
 */
async function getCatalogueTree(bookId) {
    /** @type {unknown} request 已解包的 data. */
    const data = await request.post('/v1/catalogue/tree', {book: {id: bookId}})
    if (isRecord(data) === false || isRecord(data.book) === false ||
        Array.isArray(data.book.catalogues) === false) {
        throw new Error('目录树响应缺少 book.catalogues 数组')
    }
    const book = readBookOption(data.book)
    if (book.id !== bookId) {
        throw new Error('目录树响应中的书籍 ID 与请求不一致')
    }
    return {...book, catalogues: data.book.catalogues.map(readCatalogueNode)}
}

/**
 * 保存父目录下的阅读顺序. 两个分组均包含该层全部直接子项.
 * @param {ReorderRequest} payload 已含外层 catalogue 的完整请求体.
 * @returns {Promise<void>} 排序事务已提交.
 */
async function saveCatalogueOrder(payload) {
    await request.post('/v1/catalogue/reorder', payload)
}

/**
 * 设置单个节点的目标状态, 由后端处理级联隐藏.
 * @param {string} id 目录节点 ID.
 * @param {DisplayStatus} status 目标显隐状态.
 * @returns {Promise<void>} 显隐事务已提交.
 */
async function setCatalogueDisplay(id, status) {
    await request.post('/v1/catalogue/display', {catalogue: {id, display_status: status}})
}

/**
 * 用文件节点 ID 读取文章. 响应内的 id 是文章主键.
 * @param {string} nodeId 所点击的文件节点 ID.
 * @returns {Promise<ArticlePreview>} 标题与保留换行的 Markdown 正文.
 */
async function getArticlePreview(nodeId) {
    /** @type {unknown} request 已解包的 data. */
    const data = await request.post('/v1/article/preview', {catalogue: {id: nodeId}})
    if (isRecord(data) === false || isRecord(data.article) === false) {
        throw new Error('预览响应缺少 article 对象')
    }
    return {
        id: readRequiredStringField(data.article, 'id'),
        title: readStringField(data.article, 'title'),
        content: readStringField(data.article, 'content'),
    }
}

export {getBookOptions, getCatalogueTree, saveCatalogueOrder, setCatalogueDisplay, getArticlePreview}

5. 重构书籍管理页面

按天体读取列表

进入页面先加载天体，首次保持未选状态。切换天体时清空上一份书单并发起新查询，只让最新请求更新列表、错误或加载状态。查询成功且返回空数组时，可以添加第一本书；读取失败时提供重试。标题、显隐统计和添加预览都来自当前天体的完整书单。

共用添加和编辑抽屉

添加时所属天体只读。即使 5 本书中有 2 本隐藏，新书预览仍为 6。提交的数据只取标题、显隐和当前天体 ID，两位管理员同时添加时以刷新后读到的实际序号为准。标题在去除首尾空白后按 Unicode 码点计数，最多 20 个；前端用 Array.from(title.trim()).length 与后端对齐。

浏览态只展示显隐，点击所在行的「编辑」回填表单。单独保存打开抽屉时的原值作为 expected，输入修改只写入草稿，点击保存时才构造 target。取消或关闭抽屉时，列表中的原书名和显隐仍是上次成功读取的值。

整表调序

进入调序后，保存一份原始 ID 顺序，并复制一份草稿。上移和下移只改变草稿，预览序号等于草稿行位置加 1，首行上移和末行下移禁用。书名和显隐仍随各自的 ID 展示。「保存书序」一次提交，成功后用响应中的 books 替换列表。取消调序只丢弃草稿。

“所属天体”的标题、选中值和展开列表的各个选项均居中，显隐开关旁只显示当前状态。共用下拉框组件 · 下拉框样式

书籍和目录共用上下移动按钮，首项禁用上移，末项禁用下移。共享调序按钮组件

<script setup>
import {computed, onBeforeUnmount, onMounted, reactive, ref} from 'vue'
import {ElMessage} from 'element-plus'
import {Plus, Refresh} from '@element-plus/icons-vue'
import OrderMoveButtons from '@/components/OrderMoveButtons.vue'
import AdminFilterSelect from '@/components/AdminFilterSelect.vue'
import {getBodyList} from '@/api/body'
import {addBook, getBookOptions, reorderBooks, updateBook} from '@/api/book'
import {API_CODE} from '@/constants/apiCode'
import {BOOK_TITLE_MAX_LENGTH} from '@/constants/book'
import {getApiErrorCode, getApiErrorMessage, isAuthError} from '@/utils/apiError'
import {getBodyLabel} from '@/utils/body'

/** @typedef {import('@/api/book').BookOption} BookOption */
/** @typedef {import('@/constants/book').BookDisplayStatus} BookDisplayStatus */
/** @type {import('vue').Ref<import('@/api/body').BodyRecord[]>} 已创建的天体. */
const planets = ref([])
/** 选中的天体 ID. */
const planetId = ref('')
/** @type {import('vue').Ref<BookOption[]>} 当前天体最近一次成功读取的书单. */
const books = ref([])
/** @type {import('vue').Ref<BookOption[] | null>} 独立排列的行数组, 其中的记录只读. */
const orderDraft = ref(null)
/** @type {import('vue').Ref<BookOption | null>} 编辑基线, 添加时为 null. */
const original = ref(null)
/** @type {{title: string, display_status: BookDisplayStatus}} 供抽屉编辑的副本. */
const form = reactive({title: '', display_status: 'visible'})
/** 编辑快照失效后显示最新原值, 保留表单输入. */
const reviewEdit = ref(false)
/** 抽屉和请求状态. */
const drawerOpen = ref(false)
const reading = ref(false)
const writing = ref(false)
const readError = ref('')
const titleError = ref('')
/** 每次开始读取时作废前一次结果, 包括切回同一天体的情形. */
let readVersion = 0
/** 卸载后忽略进行中的请求结果. */
let active = true

const planetLabel = computed(function () {
    const planet = planets.value.find(function (item) { return item.id === planetId.value })
    return planet === undefined ? '' : getBodyLabel(planet.name)
})
const visibleCount = computed(function () {
    return books.value.filter(function (book) { return book.display_status === 'visible' }).length
})
const orderChanged = computed(function () {
    return orderDraft.value !== null && orderDraft.value.some(function (book, index) {
        return book.id !== books.value[index].id
    })
})

/** 清除已失效的天体及相关草稿, 重新读取天体列表. */
async function recoverPlanet() {
    ElMessage.warning('所选天体已不存在, 请重新选择')
    drawerOpen.value = false
    orderDraft.value = null
    planetId.value = ''
    books.value = []
    planets.value = []
    await reload()
}

/**
 * 按当前选择读天体列表或书单. 写后刷新失败时, 错误区只提供读取操作.
 * @param {string} [failurePrefix] 用于标明之前已经提交的操作.
 * @returns {Promise<boolean>} 读取成功且结果仍属于当前页面.
 */
async function reload(failurePrefix = '') {
    const version = ++readVersion
    const id = planetId.value
    reading.value = true
    readError.value = ''
    try {
        if (id === '') {
            const result = await getBodyList()
            if (version !== readVersion) return false
            planets.value = result
        } else {
            const result = await getBookOptions(id)
            if (version !== readVersion) return false
            books.value = result
        }
        return true
    } catch (error) {
        if (version !== readVersion) return false
        if (isAuthError(error)) {
            readError.value = getApiErrorMessage(error, '请重新登录')
            return false
        }
        if (id !== '' && getApiErrorCode(error) === API_CODE.BOOK_PLANET_NOT_FOUND) {
            await recoverPlanet()
        } else {
            readError.value = failurePrefix + getApiErrorMessage(error, '读取失败, 请重试')
        }
        return false
    } finally {
        if (version === readVersion) reading.value = false
    }
}

/** @param {string} id 新选择的天体 ID. */
function selectPlanet(id) {
    if (writing.value || drawerOpen.value || orderDraft.value !== null) return
    planetId.value = id
    books.value = []
    void reload()
}

/** @param {BookOption | null} book 编辑的书籍, null 表示添加. */
function openForm(book) {
    if (writing.value || reading.value || readError.value !== '') return
    original.value = book
    form.title = book === null ? '' : book.title
    form.display_status = book === null ? 'visible' : book.display_status
    reviewEdit.value = false
    titleError.value = ''
    drawerOpen.value = true
}

/** 重新读取编辑基线. 用户的输入仍保留在 form 中, 等待再次确认. */
async function recoverEdit() {
    reviewEdit.value = true
    if (await reload('读取最新书籍失败. ') === false) return
    const latest = books.value.find(function (item) { return item.id === original.value.id })
    if (latest === undefined) {
        drawerOpen.value = false
        ElMessage.warning('书籍已不存在')
        return
    }
    original.value = latest
}

/** @param {BookOption} book 读取编辑原值, 保留历史标题的首尾空白. */
function fieldsOf(book) {
    return {title: book.title, display_status: book.display_status}
}

/**
 * 根据业务码处理写失败. 网络中断时先读取书单, 避免重复追加.
 * @param {unknown} error 写接口抛出的异常.
 */
async function handleWriteError(error) {
    if (!active || isAuthError(error)) return
    const code = getApiErrorCode(error)
    if (code === API_CODE.BOOK_PLANET_NOT_FOUND) {
        await recoverPlanet()
    } else if (code === API_CODE.BOOK_EDIT_CHANGED) {
        await recoverEdit()
    } else if (code === API_CODE.BOOK_ORDER_CHANGED || code === API_CODE.BOOK_NOT_FOUND || code === undefined) {
        drawerOpen.value = false
        orderDraft.value = null
        const message = code === undefined ? '保存结果待确认, 请核对最新书单' : getApiErrorMessage(error, '书单已变化')
        ElMessage.warning(message)
        await reload(message + '. ')
    } else {
        ElMessage.error(getApiErrorMessage(error, '保存失败'))
    }
}

/** 创建后重读完整书单. 编辑时使用后端返回的书籍摘要. */
async function submitForm() {
    if (!drawerOpen.value || writing.value || reading.value || readError.value !== '') return
    const title = form.title.trim()
    if (title === '' || Array.from(title).length > BOOK_TITLE_MAX_LENGTH) {
        titleError.value = '书籍标题需要 1 至 ' + BOOK_TITLE_MAX_LENGTH + ' 个字符'
        return
    }
    const target = {title, display_status: form.display_status}
    writing.value = true
    try {
        const before = original.value
        if (before === null) {
            await addBook({planet_id: planetId.value, ...target})
            if (!active) return
            drawerOpen.value = false
            ElMessage.success('已添加书籍')
            await reload('书籍已添加, 列表刷新失败. ')
        } else {
            const updated = await updateBook(before.id, fieldsOf(before), target)
            if (!active) return
            drawerOpen.value = false
            ElMessage.success('已保存修改')
            if (updated.sort !== before.sort) {
                await reload('修改已保存, 列表刷新失败. ')
            } else {
                books.value = books.value.map(function (book) { return book.id === updated.id ? updated : book })
            }
        }
    } catch (error) {
        await handleWriteError(error)
    } finally {
        if (active) writing.value = false
    }
}

/** 复制列表顺序. 调序期间锁定读取和编辑入口, 保留 books 作为原始顺序. */
function beginOrder() {
    if (writing.value || reading.value || books.value.length < 2) return
    orderDraft.value = books.value.slice()
}

/**
 * @param {number} index 行下标.
 * @param {number} step 单次移动量.
 */
function move(index, step) {
    const rows = orderDraft.value
    const next = index + step
    if (writing.value || rows === null || next < 0 || next >= rows.length) return
    const item = rows.splice(index, 1)[0]
    rows.splice(next, 0, item)
}

/**
 * @param {BookOption} book 书籍摘要.
 * @returns {string} 后端返回的字符串 ID.
 */
function bookId(book) {
    return book.id
}

/** 一次提交完整快照和目标, 以响应中的完整书单结束调序. */
async function saveOrder() {
    if (writing.value || orderDraft.value === null || !orderChanged.value) return
    writing.value = true
    try {
        const result = await reorderBooks(planetId.value, books.value.map(bookId), orderDraft.value.map(bookId))
        if (!active) return
        orderDraft.value = null
        books.value = result
        ElMessage.success('已保存书序')
    } catch (error) {
        await handleWriteError(error)
    } finally {
        if (active) writing.value = false
    }
}

onMounted(function () { void reload() })
onBeforeUnmount(function () {
    active = false
    readVersion++
})
</script>

结果页面应怎样恢复
创建成功，回读书单失败写入已经确认，提示「已添加，书单刷新失败」。刷新动作只发 options，避免把同一本书再添加一次。
编辑返回 20204当前书名或显隐已与打开抽屉时不同。读取最新书单，对照自己的草稿与服务器原值，重新确认后才以最新快照提交。仅调整了书序不构成编辑冲突。
调序返回 20203书籍成员或顺序已变化。重新读取完整书单，在最新顺序上再次调整。另一人仅编辑书名或显隐时，可以照常提交调序；成功响应会带回最新书籍字段。
20102 / 20202所选天体失效时清空上下文并刷新天体选项。编辑的书已不存在时结束该条编辑，重读所选天体的书单。
10101检查标题、枚举、字符串 ID 和请求层级。调序的两份数组必须包含相同的一组书籍，每本恰好一次，包含隐藏书籍。与当前数据库顺序不符则属于 20203。
超时、连接中断如果写响应没有抵达，结果可能已提交。先查书单核对，再由管理员决定是否重新操作，尤其要避免自动重发创建请求。
20003 / 20004沿用公共鉴权处理。页面通过 isAuthError() 识别已由拦截器处理的异常。

公共错误文案沿用 getApiErrorMessage()。业务码 500 也可能由其他内部异常引起；只能结合后端日志确认是否存在历史书序异常。公共错误处理 · 历史书序检查与整理说明
